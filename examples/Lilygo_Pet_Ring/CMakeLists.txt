cmake_minimum_required(VERSION 3.16)
project(lilygo_pet_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(pet_tests tests/pet_tests.cpp PetModel.cpp PixelRenderer.cpp PetLocale.cpp)
if(MSVC)
  target_compile_options(pet_tests PRIVATE /W4)
else()
  target_compile_options(pet_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME pet_tests COMMAND pet_tests)
foreach(test_name initial time-chunks growth care save clock render motion room ground branches needs health migration locale companion)
  add_test(NAME pet_${test_name} COMMAND pet_tests ${test_name})
endforeach()
add_executable(render_preview tools/render_preview.cpp PetModel.cpp PixelRenderer.cpp PetLocale.cpp)
add_executable(petlink_core_tests ../../libraries/PetLink/tests/core_tests.cpp)
target_include_directories(petlink_core_tests PRIVATE ../../libraries/PetLink/src)
if(NOT MSVC)
  target_compile_options(petlink_core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME petlink_core COMMAND petlink_core_tests)
